export function compareDoorJourneys(query: MobilityQuery, a: DoorJourney, b: DoorJourney) {
  const arrival = Date.parse(a.arrivalTime) - Date.parse(b.arrivalTime), departure = Date.parse(b.departureTime) - Date.parse(a.departureTime)
  const duration = a.durationSeconds - b.durationSeconds, transfers = a.transit.transfers - b.transit.transfers
  switch (query.order) {
    case 'duration': return duration || arrival || transfers
    case 'transfers': return transfers || (query.mode === 'arrival' ? departure : arrival) || duration
    case 'walking': return Number(a.unknownTransferWalking) - Number(b.unknownTransferWalking) || a.walkingSeconds - b.walkingSeconds || arrival
    case 'waiting': return Number(a.waitingSeconds === null) - Number(b.waitingSeconds === null) || (a.waitingSeconds ?? Infinity) - (b.waitingSeconds ?? Infinity) || arrival
    case 'departure': return departure || duration || transfers
    default: return arrival || duration || transfers
  }
}
export function selectDoorJourneys(query: MobilityQuery, choices: DoorJourney[]): DoorJourney[] {
  const seen = new Set<string>()
  const unique = choices.filter(item => { const id = JSON.stringify(item.transit.legs.map(leg => [leg.tripId, leg.serviceDate, leg.from.id, leg.to.id])); if (seen.has(id)) return false; seen.add(id); return true })
  const dominates = (a: DoorJourney, b: DoorJourney) => !a.unknownTransferWalking && a.waitingSeconds !== null &&
    (b.waitingSeconds !== null ? a.waitingSeconds <= b.waitingSeconds : query.order !== 'waiting') &&
    Date.parse(a.departureTime) >= Date.parse(b.departureTime) && Date.parse(a.arrivalTime) <= Date.parse(b.arrivalTime) && a.durationSeconds <= b.durationSeconds && a.transit.transfers <= b.transit.transfers && a.walkingSeconds <= b.walkingSeconds &&
    (a.durationSeconds < b.durationSeconds || a.transit.transfers < b.transit.transfers || a.walkingSeconds < b.walkingSeconds || b.waitingSeconds !== null && a.waitingSeconds < b.waitingSeconds)
  const ranked = unique.filter(item => !unique.some(other => other !== item && dominates(other, item))).sort((a, b) => compareDoorJourneys(query, a, b))
  const patterns = new Set<string>(), selected: DoorJourney[] = []
  for (const item of ranked) {
    const pattern = JSON.stringify(item.transit.legs.map(leg => [leg.routeId, leg.directionId, leg.from.id, leg.to.id]))
    if (!patterns.has(pattern) && selected.length < 6) { patterns.add(pattern); selected.push(item) }
  }
  for (const item of ranked) if (selected.length < 6 && !selected.includes(item)) selected.push(item)
  return selected.sort((a, b) => compareDoorJourneys(query, a, b))
}
export function doorJourney(_query: MobilityQuery, transitResult: DoorJourney['transitResult'], transit: DoorJourney['transit'], access: WalkPath | null, egress: WalkPath | null, fromName: string, toName: string): DoorJourney {
  const safetySeconds = 0
  const departure = Date.parse(transit.departureTime) - ((access?.durationSeconds || 0) + safetySeconds) * 1000
  const arrival = Date.parse(transit.arrivalTime) + (egress?.durationSeconds || 0) * 1000
  const connections = transit.connections ?? (transit.connection ? [transit.connection] : [])
  const walkingSeconds = (access?.durationSeconds || 0) + (egress?.durationSeconds || 0) + connections.reduce((sum, connection) => sum + (connection.walkingSeconds || 0), 0)
  const unknownTransferWalking = connections.some(connection => connection.walkingSeconds === null)
  return { transit, transitResult, access, egress, fromName, toName, safetySeconds, departureTime: new Date(departure).toISOString(), arrivalTime: new Date(arrival).toISOString(), durationSeconds: (arrival - departure) / 1000, walkingSeconds, unknownTransferWalking, waitingSeconds: unknownTransferWalking ? null : safetySeconds + connections.reduce((sum, connection) => sum + connection.intervalSeconds - (connection.walkingSeconds || 0), 0) }
}

      }
      await reportStage(progress, 'ranking')

      metrics.rankingMs = performance.now() - rankingStart
      if (limited) warnings.push('Budget GTFS atteint : comparaison partielle, les propositions affichées restent contrôlées.')

      warnings.push('Classement parmi les propositions retenues uniquement, pas de garantie d’optimalité globale. Les attentes non séparables du minimum GTFS restent inconnues.')
      if ((from.address && !origins.length) || (to.address && !destinations.length)) warnings.push('Aucun accès piéton utilisable confirmé pour une adresse. Aucun remplacement à vol d’oiseau.')
      signal.throwIfAborted()
      metrics.durationMs = Math.round((performance.now() - started) * 100) / 100
      Object.assign(metrics, { geocodingMs: trace.geocodingMs, walkingMs: trace.walkingMs, upstreamRequests: trace.upstreamRequests, cacheHits: trace.cacheHits, queueMs: trace.queueMs })
      return { state: limited || metrics.walkBudgetReached ? 'limited' : itineraries.length ? 'ok' : outside ? 'outside_validity' : 'no_journey', itineraries, warnings: [...new Set(warnings)], metrics, providerIssues }
    },
  }
}

import type { GeographicProvider } from './provider.ts'
import { createIgnProvider } from './provider.ts'
import { createPhotonProvider } from './photon.ts'
import { createValhallaProvider } from './valhalla.ts'

export function createHybridGeographicProvider(): GeographicProvider {
  const ign = createIgnProvider()
  const photon = createPhotonProvider()
  const valhalla = createValhallaProvider()
  return {
    addresses: (q, trace, signal) => photon.addresses(q, trace, signal),
    walk: (from, to, trace, signal) => valhalla.walk(from, to, trace, signal),
    resolveAddress: async (id, name, trace, signal) =>
      id.startsWith('address:photon:')
        ? photon.resolveAddress(id, name, trace, signal)
        : id.startsWith('address:')
          ? ign.resolveAddress?.(id, name, trace, signal)
          : Promise.resolve(undefined)
  }
}

import type { AddressPlace, Point, WalkPath } from '../../src/types/mobility.ts'
import { createSharedSearch } from '../search/shared.ts'

export class GeographicError extends Error {}
export const distance = (a: Point, b: Point) => {
  const rad = Math.PI / 180, lat = (a.latitude + b.latitude) / 2 * rad
  return Math.hypot((a.latitude - b.latitude) * rad, (a.longitude - b.longitude) * rad * Math.cos(lat)) * 6371000
}
export interface GeographicProvider {
  addresses(q: string, trace?: GeographicTrace, signal?: AbortSignal): Promise<AddressPlace[]>
  walk(from: Point, to: Point, trace?: GeographicTrace, signal?: AbortSignal): Promise<WalkPath>
  resolveAddress?(id: string, name: string, trace?: GeographicTrace, signal?: AbortSignal): Promise<AddressPlace | undefined>
}
export interface GeographicTrace { geocodingMs: number; walkingMs: number; queueMs: number; upstreamRequests: number; cacheHits: number; coalesced: number }
export const geographicTrace = (): GeographicTrace => ({ geocodingMs: 0, walkingMs: 0, queueMs: 0, upstreamRequests: 0, cacheHits: 0, coalesced: 0 })
/** One bounded queue per backend process. Never switches profile or provider on failure. */
export function createIgnProvider(fetcher: typeof fetch = fetch, now = () => Date.now()): GeographicProvider {
  const cache = new Map<string, { until: number; value: unknown }>()
  const pending = createSharedSearch<unknown>(64)
  const addressIds = new Map<string, { until: number; value: AddressPlace }>()
  let gate: Promise<unknown> = Promise.resolve(), blockedUntil = new Map<string, number>(), active = 0
  const waiters: (() => void)[] = []
  async function slot() {
    if (active >= 2) await new Promise<void>(resolve => waiters.push(resolve))
    else active++
    const turn = gate.catch(() => {}).then(() => new Promise(resolve => setTimeout(resolve, 260)))
    gate = turn
    await turn
  }
  function release() { const next = waiters.shift(); if (next) next(); else active-- }
  async function load<T>(url: URL, ttl: number, parse: (data: unknown) => T, trace?: GeographicTrace, consumerSignal: AbortSignal = new AbortController().signal): Promise<T> {
    consumerSignal.throwIfAborted()
    const key = url.href, hit = cache.get(key)
    if (hit && hit.until > now()) { if (trace) trace.cacheHits++; return hit.value as T }
    if (pending.has(key) && trace) trace.coalesced++
    try { return await pending.run(key, async upstreamSignal => {
      const waiting = performance.now()
      await slot()
      if (trace) trace.queueMs += performance.now() - waiting
      try {
      upstreamSignal.throwIfAborted()
      if ((blockedUntil.get(url.pathname) ?? 0) > now()) throw new GeographicError('Service IGN temporairement limité. Réessayez plus tard.')
      const networkStart = performance.now()
      if (trace) trace.upstreamRequests++
      let response: Response
      try {
      response = await fetcher(url, { signal: AbortSignal.any([upstreamSignal, AbortSignal.timeout(8000)]), redirect: 'error', headers: { Accept: 'application/json' } })
      if ([429, 502, 503, 504].includes(response.status)) {
        const retry = response.headers.get('Retry-After')
        const seconds = retry && /^\d+$/.test(retry) ? Number(retry) * 1000 : retry ? Date.parse(retry) - now() : 30000
        blockedUntil.set(url.pathname, now() + Math.max(1000, Number.isFinite(seconds) ? seconds : 30000))
      }
      if (!response.ok) { await response.body?.cancel(); throw new GeographicError(`Service IGN indisponible (HTTP ${response.status}).`) }
      const text = await response.text()
      if (text.length > 1000000) throw new GeographicError('Réponse IGN trop volumineuse.')
      const value = parse(JSON.parse(text))
      upstreamSignal.throwIfAborted()
      if (cache.size >= 256) cache.delete(cache.keys().next().value!)
      cache.set(key, { until: now() + ttl, value }); return value
      } finally { if (trace) trace[url.pathname.includes('geocodage') ? 'geocodingMs' : 'walkingMs'] += performance.now() - networkStart }
      } finally { release() }
    }, consumerSignal) as T }
    catch (error) { consumerSignal.throwIfAborted(); throw error instanceof GeographicError ? error : new GeographicError('Service IGN inaccessible ou réponse invalide. Aucun parcours piéton inventé.') }
  }
  const provider: GeographicProvider = {
    addresses(q, trace, signal) {
      if (q.trim().length < 3 || q.length > 160) return Promise.resolve([])
      const url = new URL('https://data.geopf.fr/geocodage/search')
      url.search = new URLSearchParams({ q, limit: '6', indexes: 'address', lat: '45.75', lon: '4.85', autocomplete: '1' }).toString()
      return load(url, 300000, raw => {
        const data = raw as { features?: { geometry?: { type: string; coordinates: number[] }; properties?: Record<string, unknown> }[] }
        if (!Array.isArray(data.features)) throw new Error('Invalid geocoding')
        const results = data.features.flatMap(feature => {
          const p = feature.properties, c = feature.geometry?.coordinates
          if (!p || feature.geometry?.type !== 'Point' || !c || c.length !== 2 || !c.every(Number.isFinite) || Math.abs(c[0]) > 180 || Math.abs(c[1]) > 90 || typeof p.id !== 'string' || typeof p.label !== 'string' || !['housenumber', 'street'].includes(String(p.type))) return []
          return [{ kind: 'address' as const, id: `address:${p.id}`, name: p.label, longitude: c[0], latitude: c[1], precision: p.type as AddressPlace['precision'], source: 'BAN via IGN' as const }]
        })
        for (const value of results) { if (addressIds.size >= 256) addressIds.delete(addressIds.keys().next().value!); addressIds.set(value.id, { until: now() + 300000, value }) }
        return results
      }, trace, signal)
    },
    walk(from, to, trace, signal) {
      const url = new URL('https://data.geopf.fr/navigation/itineraire')
      url.search = new URLSearchParams({ resource: 'bdtopo-osrm', start: `${from.longitude},${from.latitude}`, end: `${to.longitude},${to.latitude}`, profile: 'pedestrian', optimization: 'fastest', geometryFormat: 'geojson', distanceUnit: 'meter', timeUnit: 'second', crs: 'EPSG:4326', getSteps: 'false' }).toString()
      return load(url, 3600000, raw => {
        const data = raw as { profile?: string; resource?: string; resourceVersion?: string; crs?: string; timeUnit?: string; distanceUnit?: string; duration?: number; distance?: number; geometry?: WalkPath['geometry'] }
        const points = data.geometry?.coordinates
        if (data.profile !== 'pedestrian' || data.resource !== 'bdtopo-osrm' || data.crs !== 'EPSG:4326' || data.timeUnit !== 'second' || data.distanceUnit !== 'meter' || !Number.isFinite(data.duration) || data.duration! <= 0 || data.duration! > 3600 || !Number.isFinite(data.distance) || data.distance! <= 0 || data.distance! > 4000 || data.geometry?.type !== 'LineString' || !Array.isArray(points) || points.length < 2 || points.length > 20000 || points.some(p => !Array.isArray(p) || p.length !== 2 || !p.every(Number.isFinite) || Math.abs(p[0]) > 180 || Math.abs(p[1]) > 90)) throw new Error('Invalid pedestrian response')
        const point = (p: number[]): Point => ({ longitude: p[0], latitude: p[1] })
        const startSnapMeters = distance(from, point(points[0])), endSnapMeters = distance(to, point(points.at(-1)!))
        // Reject network snaps that would silently omit substantial access walking.
        if (startSnapMeters > 25 || endSnapMeters > 25) throw new GeographicError('Accès au réseau piéton trop éloigné du point demandé ; parcours exclu.')
        return { source: 'IGN BD TOPO', profile: 'pedestrian', resourceVersion: data.resourceVersion || 'non renseignée', fetchedAt: new Date(now()).toISOString(), durationSeconds: Math.ceil(data.duration!), distanceMeters: data.distance!, startSnapMeters, endSnapMeters, geometry: data.geometry! }
      }, trace, signal)
    },
    async resolveAddress(id, name, trace, signal) {
      signal?.throwIfAborted()
      const hit = addressIds.get(id)
      if (hit && hit.until > now() && hit.value.name === name) { if (trace) trace.cacheHits++; return hit.value }
      return (await provider.addresses(name, trace, signal)).find(value => value.id === id)
    },
  }
  return provider
}

export interface RoutingLimits {
  maxEvents: number
  maxLabels: number
  maxLabelsPerStop: number
  maxBoardStops: number
  maxWalkRequests: number
  maxAccessSeconds: number
  radiusMeters: number
  deadlineMs: number
}
export const defaultRoutingLimits: RoutingLimits = { maxEvents: 240000, maxLabels: 50000, maxLabelsPerStop: 64, maxBoardStops: 1600, maxWalkRequests: 48, maxAccessSeconds: 1200, radiusMeters: 1200, deadlineMs: 45000 }
export function routingLimits(env: Record<string, unknown> = {}): RoutingLimits {
  const result = { ...defaultRoutingLimits }
  for (const key of Object.keys(result) as (keyof RoutingLimits)[]) {
    const name = 'ROUTING_' + key.replace(/[A-Z]/g, letter => '_' + letter.toLowerCase()).toUpperCase()
    if (env[name] !== undefined) {
      const value = Number(env[name])
      if (!Number.isInteger(value) || value < 1 || value > defaultRoutingLimits[key] * 4) throw new Error(`Configuration ${name} invalide.`)
      result[key] = value
    }
  }
  return result
}

interface Path { legs: ScheduledLeg[]; connection: JourneyTransfer | null }
const duration = (path: Path) => path.legs.at(-1)!.arrival - path.legs[0].departure
function comparison(query: RoutingQuery) {
  return (a: Path, b: Path) => {
    const arrival = a.legs.at(-1)!.arrival - b.legs.at(-1)!.arrival
    const departure = b.legs[0].departure - a.legs[0].departure
    const length = duration(a) - duration(b)
    const transfers = a.legs.length - b.legs.length
    switch (query.order) {
      case 'duration': return length || transfers || arrival || departure
      case 'transfers': return transfers || (query.mode === 'arrival' ? departure : arrival) || length
      case 'departure': return departure || length || transfers || arrival
      default: return arrival || length || transfers || departure
    }
  }
}
function lowerBound(legs: ScheduledLeg[], time: number) {
  let lo = 0, hi = legs.length
  while (lo < hi) { const mid = (lo + hi) >>> 1; if (legs[mid].departure < time) lo = mid + 1; else hi = mid }
  return lo
}
function publicLeg(leg: ScheduledLeg, topology: Topology): RideLeg {
  const line = topology.routes.get(leg.routeId)!
  return {
    tripId: leg.tripId, serviceDate: leg.serviceDate, routeId: leg.routeId,
    lineName: line.short_name || line.long_name, routeType: line.type, directionId: leg.directionId, headsign: leg.headsign,
    from: { id: leg.fromId, name: topology.stops.get(leg.fromId)!.name, sequence: leg.fromSequence },
    to: { id: leg.toId, name: topology.stops.get(leg.toId)!.name, sequence: leg.toSequence },
    departureTime: new Date(leg.departure).toISOString(), arrivalTime: new Date(leg.arrival).toISOString(),
    departureGtfsSeconds: leg.departureSeconds, arrivalGtfsSeconds: leg.arrivalSeconds,
    interpolatedBySource: leg.fromTimepoint === 0 || leg.toTimepoint === 0,
  }
}

/** Moteur borné : directs et une correspondance, élagage topologique puis
 * jointures temporelles indexées. Ce n'est pas un RAPTOR réseau complet.
 */
function createLegacyJourneyPlanner(db: QueryDatabase) {
  let cachedTopology: Promise<Topology> | null = null
  let cachedHash = ''
  return {
    async search(query: RoutingQuery, pedestrianTransfers: VerifiedTransfer[] = [], progress?: ProgressObserver): Promise<RoutingResult> {
      await reportStage(progress, 'connections')
      if (!Number.isInteger(query.marginSeconds) || query.marginSeconds < 0 || query.marginSeconds > 1800) throw new Error('La marge doit être comprise entre 30 et 1800 secondes.')
      const started = performance.now()
      let sqlQueries = 0
      const measured: QueryDatabase = { all: async <T>(sql: string, values?: (string | number | null)[]) => { sqlQueries++; return db.all<T>(sql, values) } }
      const metadata = await getMetadata(measured)
      const coldTopology = !cachedTopology || cachedHash !== metadata.sha256
      if (coldTopology) {
        cachedHash = metadata.sha256
        cachedTopology = loadTopology(measured).catch(error => { cachedTopology = null; throw error })
      }
      const topology = await cachedTopology!
      const walks = new Map(pedestrianTransfers.map(edge => [JSON.stringify([edge.from, edge.to]), edge.path]))
      const edges = new Map(topology.edges)
      for (const edge of pedestrianTransfers) if (!edges.has(JSON.stringify([edge.from, edge.to]))) edges.set(JSON.stringify([edge.from, edge.to]), { from: edge.from, to: edge.to })
      for (const id of [query.fromStopId, query.toStopId]) if (topology.stops.get(id)?.location_type !== 0) throw new Error('Sélectionnez des quais GTFS existants.')
      if (query.fromStopId === query.toStopId) throw new Error('Le départ et l’arrivée doivent être différents.')
      const at = Date.parse(query.at)
      const low = query.mode === 'departure' ? at : at - 6 * 3600000
      const high = query.mode === 'departure' ? at + 6 * 3600000 : at
      const scope = { maxTransfers: 1 as const, windowHours: 6 as const, from: new Date(low).toISOString(), until: new Date(high).toISOString(), marginSeconds: 0 }
      const metrics = { durationMs: 0, sqlQueries: 0, legCount: 0, transferPairs: 0, candidateCount: 0, coldTopology }
      const warnings = [
        'Périmètre : trajets directs ou une correspondance, dans une fenêtre de six heures. Pas de marche d’accès ou de sortie.',
        `Minima GTFS et marche confirmée, sans marge supplémentaire.`,
        'Les changements de quai sans lien orienté et durée minimale GTFS sont exclus. Le temps de marche seul n’est pas inventé.',
        'Sélection diversifiée de six propositions au maximum ; les perturbations et le temps réel ne sont pas pris en compte.',
      ]
      if (pedestrianTransfers.length) {
        warnings[2] = 'Changements de quai limités aux liens GTFS ou aux parcours piétons IGN fournis ; aucune durée à vol d’oiseau.'
        warnings.push('Liens piétons supplémentaires calculés par IGN sur le réseau BD TOPO ; durées estimées. Les interdictions GTFS restent prioritaires.')
      }
      const finish = (state: RoutingResult['state'], itineraries: TheoreticalJourney[] = []): RoutingResult => {
        metrics.durationMs = Math.round((performance.now() - started) * 100) / 100
        metrics.sqlQueries = sqlQueries
        return { state, itineraries, query, metadata, scope, metrics, warnings }
      }
      const days = serviceDays(metadata, low, high)
      if (!days.length) return finish('outside_validity')
      const fromRoutes = topology.stopRoutes.get(query.fromStopId) || new Set<string>()
      const toRoutes = topology.stopRoutes.get(query.toStopId) || new Set<string>()
      const intersection = (routes: Set<string> | undefined, wanted: Set<string>) => [...routes || []].filter(id => wanted.has(id))
      const directRoutes = intersection(fromRoutes, toRoutes)
      const pairs: LegPair[] = []
      const pairIndices = new Map<string, number>()
      function addPair(from: string, to: string, routes: string[]) {
        const key = JSON.stringify([from, to])
        const existing = pairIndices.get(key)
        if (existing !== undefined) { pairs[existing].routes = [...new Set([...pairs[existing].routes, ...routes])]; return existing }
        const index = pairs.push({ from, to, routes }) - 1
        pairIndices.set(key, index); return index
      }
      const direct = directRoutes.length ? addPair(query.fromStopId, query.toStopId, directRoutes) : -1
      const connections: { from: string; to: string; first: number; second: number }[] = []
      for (const edge of edges.values()) {
        if ([query.fromStopId, query.toStopId].includes(edge.from) || [query.fromStopId, query.toStopId].includes(edge.to)) continue
        const first = intersection(topology.stopRoutes.get(edge.from), fromRoutes)
        const second = intersection(topology.stopRoutes.get(edge.to), toRoutes)
        if (!first.length || !second.length) continue
        connections.push({ ...edge, first: addPair(query.fromStopId, edge.from, first), second: addPair(edge.to, query.toStopId, second) })
      }
      metrics.transferPairs = connections.length
      if (connections.length > 256) { warnings.push('Limite de 256 liens de correspondance atteinte : recherche refusée pour ne pas annoncer un classement incomplet.'); return finish('limited') }
      const legs = await scheduledLegs(measured, pairs, days, low, high)
      metrics.legCount = legs.length
      if (legs.length > 20000) { warnings.push('Plus de 20 000 tronçons candidats : réduisez le périmètre. Aucun résultat tronqué n’est présenté.'); return finish('limited') }
      const byPair = new Map<number, ScheduledLeg[]>()
      for (const leg of legs) { const group = byPair.get(leg.pairIndex) || []; group.push(leg); byPair.set(leg.pairIndex, group) }
      const compare = comparison(query)
      const bestPatterns = new Map<string, Path>()
      let bestSchedules: Path[] = []
      const directLegs = byPair.get(direct) || []
      const dominates = (a: Path, b: Path) => a.legs[0].departure >= b.legs[0].departure && a.legs.at(-1)!.arrival <= b.legs.at(-1)!.arrival && a.legs.length <= b.legs.length &&
        (a.legs[0].departure > b.legs[0].departure || a.legs.at(-1)!.arrival < b.legs.at(-1)!.arrival || a.legs.length < b.legs.length)
      function consider(path: Path) {
        metrics.candidateCount++
        if (path.legs.length > 1 && directLegs.some(leg => leg.departure >= path.legs[0].departure && leg.arrival <= path.legs.at(-1)!.arrival)) return
        if (bestSchedules.some(existing => dominates(existing, path)) || [...bestPatterns.values()].some(existing => dominates(existing, path))) return
        for (const [key, existing] of bestPatterns) if (dominates(path, existing)) bestPatterns.delete(key)
        const pattern = JSON.stringify(path.legs.map(leg => [leg.routeId, leg.fromId, leg.toId, leg.directionId]))
        const previous = bestPatterns.get(pattern)
        if (!previous || compare(path, previous) < 0) bestPatterns.set(pattern, path)
        bestSchedules = bestSchedules.filter(existing => !dominates(path, existing))
        bestSchedules.push(path)
        bestSchedules.sort(compare)
        bestSchedules = bestSchedules.slice(0, 6)
      }
      for (const leg of directLegs) consider({ legs: [leg], connection: null })
      let checks = 0
      for (const edge of connections) {
        const arriving = byPair.get(edge.first) || []
        const departing = byPair.get(edge.second) || []
        for (const first of arriving) {
          for (let index = lowerBound(departing, first.arrival); index < departing.length; index++) {
            if (++checks > 200000) { warnings.push('Budget de combinaisons dépassé : aucun résultat partiel présenté.'); return finish('limited') }
            const second = departing[index]
            if (first.tripId === second.tripId && first.serviceDate === second.serviceDate) continue
            const walk = walks.get(JSON.stringify([edge.from, edge.to]))
            const policy = transferPolicy(topology, edge.from, edge.to, first, second, walk?.durationSeconds)
            if (!policy) continue
            const required = policy.minimum
            const interval = (second.departure - first.arrival) / 1000
            if (interval < required) continue
            consider({ legs: [first, second], connection: {
              fromStopId: edge.from, fromStopName: topology.stops.get(edge.from)!.name,
              toStopId: edge.to, toStopName: topology.stops.get(edge.to)!.name,
              ruleId: policy.ruleId, source: policy.source, officialMinimumSeconds: policy.officialMinimum,
              marginSeconds: 0, requiredSeconds: required,
              intervalSeconds: interval, spareSeconds: interval - required, walkingSeconds: policy.walking,
              ...(policy.walking !== null && policy.walking > 0 && walk ? { walkingPath: walk } : {}),
            } })
          }
        }
      }
      await reportStage(progress, 'ranking')
      const selected: Path[] = [...bestPatterns.values()].sort(compare).slice(0, 6)
      const identity = (path: Path) => JSON.stringify(path.legs.map(leg => [leg.tripId, leg.serviceDate, leg.fromSequence, leg.toSequence]))
      const seen = new Set(selected.map(identity))
      for (const path of bestSchedules) if (selected.length < 6 && !seen.has(identity(path))) { selected.push(path); seen.add(identity(path)) }
      selected.sort(compare)
      const itineraries = selected.map((path, index) => ({
        id: `proposal-${index + 1}`, kind: 'theoretical' as const, departureTime: new Date(path.legs[0].departure).toISOString(),
        arrivalTime: new Date(path.legs.at(-1)!.arrival).toISOString(), durationSeconds: duration(path) / 1000,
        initialWaitSeconds: query.mode === 'departure' ? (path.legs[0].departure - at) / 1000 : 0,
        transfers: path.legs.length - 1, legs: path.legs.map(leg => publicLeg(leg, topology)), connection: path.connection,
      }))
      return finish(itineraries.length ? 'ok' : 'no_journey', itineraries)
    },
  }
}
export function createJourneyPlanner(db: QueryDatabase, limits?: RoutingLimits) {
  const legacy = createLegacyJourneyPlanner(db)
  const network = createNetworkPlanner(db, limits)
  return {
    search: (query: RoutingQuery, pedestrian: VerifiedTransfer[] = [], seeds?: NetworkSeeds, signal?: AbortSignal, progress?: ProgressObserver) => query.maxTransfers === 2 || seeds ? network.search(query, pedestrian, seeds, signal, progress) : legacy.search(query, pedestrian, progress),
  }
}
export type JourneyPlanner = ReturnType<typeof createJourneyPlanner>

import type { ProgressObserver, SearchProgress } from '../../src/types/searchProgress.ts'

interface Subscriber { observer?: ProgressObserver }
interface Job<T> { controller: AbortController; subscribers: Set<Subscriber>; promise: Promise<T>; last?: SearchProgress }
/** In-flight only, scoped by the API to one immutable DB binding. No address log. */
export function createSharedSearch<T>(capacity = 8) {
  const jobs = new Map<string, Job<T>>()
  return {
    has: (key: string) => jobs.has(key),
    run(key: string, work: (signal: AbortSignal, observer: ProgressObserver) => Promise<T>, signal: AbortSignal, observer?: ProgressObserver): Promise<T> {
      signal.throwIfAborted()
      let job = jobs.get(key)
      if (!job) {
        if (jobs.size >= capacity) return Promise.reject(new Error('Le calculateur est occupé. Réessayez dans un instant.'))
        const created: Job<T> = { controller: new AbortController(), subscribers: new Set(), promise: Promise.resolve(undefined as T) }
        const cleanup = () => { if (jobs.get(key) === created) jobs.delete(key) }
        created.promise = Promise.resolve().then(() => {
          created.controller.signal.throwIfAborted()
          return work(created.controller.signal, event => {
            created.last = event
            for (const subscriber of created.subscribers) { try { subscriber.observer?.(event) } catch { /* A disconnected listener cannot corrupt a shared result. */ } }
          })
        }).then(value => { cleanup(); return value }, error => { cleanup(); throw error })
        jobs.set(key, created); job = created
      }
      const entry = job, subscriber = { observer }
      entry.subscribers.add(subscriber)
      if (entry.last) observer?.(entry.last)
      return new Promise<T>((resolve, reject) => {
        let detached = false
        const detach = () => { if (detached) return; detached = true; signal.removeEventListener('abort', abort); entry.subscribers.delete(subscriber) }
        const abort = () => {
          detach(); reject(signal.reason)
          if (!entry.subscribers.size) { entry.controller.abort(); if (jobs.get(key) === entry) jobs.delete(key) }
        }
        signal.addEventListener('abort', abort, { once: true })
        entry.promise.then(value => { if (!detached) { detach(); resolve(value) } }, error => { if (!detached) { detach(); reject(error) } })
        if (signal.aborted) abort()
      })
    },
  }
}

import type { ProgressObserver } from '../../src/types/searchProgress.ts'

export function wantsSearchStream(request: Request) { return request.headers.get('Accept')?.includes('application/x-ndjson') === true }
export function searchStream(request: Request, operation: (signal: AbortSignal, observer: ProgressObserver) => Promise<Response>): Response {
  const controller = new AbortController(), encoder = new TextEncoder()
  const signal = AbortSignal.any([request.signal, controller.signal])
  const body = new ReadableStream<Uint8Array>({
    start(stream) {
      const abort = () => stream.error(signal.reason)
      signal.addEventListener('abort', abort, { once: true })
      const send = (data: unknown) => { if (!signal.aborted) stream.enqueue(encoder.encode(JSON.stringify(data) + '\n')) }
      void (async () => {
      try {
        const response = await operation(signal, event => send({ type: 'stage', ...event }))
        signal.throwIfAborted()
        const value = await response.json()
        if (response.ok) send({ type: 'result', value })
        else send({ type: 'error', message: value.error || 'Calculateur indisponible.' })
      } catch { if (!signal.aborted) send({ type: 'error', message: 'Le calculateur est indisponible ou le délai est dépassé.' }) }
      finally { signal.removeEventListener('abort', abort); if (!signal.aborted) stream.close() }
      })()
    },
    cancel() { controller.abort() },
  })
  return new Response(body, { headers: { 'Content-Type': 'application/x-ndjson; charset=utf-8', 'Cache-Control': 'no-store, no-transform', 'X-Accel-Buffering': 'no' } })
}
